38:     pub fn add_nurbssurface(&mut self, surface: &session_rust::NurbsSurface, device: &wgpu::Device, queue: &wgpu::Queue) {
39:         let guid = surface.guid().to_string();
40:         let mut mesh = session_rust::remesh_nurbssurface_grid::RemeshNurbsSurfaceGrid::from_u_v(
41:             surface.clone(), 32, 32);
42:         let (vs, is) = mesh_to_vertices(&mesh);
43:         if vs.is_empty() { return; }
44:         let instance_id = self.pick.allocate(&guid);
45:         self.tri.allocate(&guid, &vs, Some(&is), instance_id, device, queue);
46:         // Boundary edges: 4 iso-curves.
47:         // iso_curve(0,v) = u-direction curve at fixed v â†’ use v-domain (dir=1) for parameter.
48:         // iso_curve(1,u) = v-direction curve at fixed u â†’ use u-domain (dir=0) for parameter.
49:         let mut edge_segs: Vec<CylinderSegment> = Vec::new();
50:         let domains = [surface.domain(0), surface.domain(1)];
51:         for (iso_dir, param_dir, t_is_max) in [(0usize, 1usize, false), (0, 1, true), (1, 0, false), (1, 0, true)] {
52:             let t = if let Some((t0, t1)) = domains[param_dir] {
53:                 if t_is_max { t1 } else { t0 }
54:             } else { continue };
55:             if let Some(crv) = surface.iso_curve(iso_dir, t) {
56:                 let (pts, _) = crv.to_polyline_adaptive(
57:                     session_rust::Tolerance::ANGULARDEFLECTION * 0.1, 0.0, 0.0);
58:                 edge_segs.extend(pts_to_segments(&pts, instance_id));
59:             }
60:         }
61:         if !edge_segs.is_empty() {
62:             if let Some(lc) = surface.linecolors.get(0) {
63:                 let c = color_to_rgba_f32(lc);
64:                 for seg in &mut edge_segs { seg.color = c; }
65:             }
66:             let seg_start = self.segments_cpu.len();
67:             self.segments_cpu.extend(edge_segs);
68:             self.guid_to_seg.insert(guid.clone(), seg_start..self.segments_cpu.len());
69:             self.segments_dirty = true;
70:         }
71:         mesh.ensure_triangle_bvh();
72:         self.nurbs_pick_meshes.insert(guid.clone(), mesh);
73:         self.nurbs_surfaces.insert(guid.clone(), surface.clone());
74:         let surf_color = color_to_rgba_f32(surface.facecolors.get(0).unwrap_or(&session_rust::Color::white()));
75:         self.write_instance_flags(instance_id, surf_color, InstanceData::FLAG_SMOOTH, device, queue);
76:     }
